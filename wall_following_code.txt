// micromouse_handturn.ino
// Hand on left sensor = left 90, right sensor = right 90, front sensor = forward
// Needs the "VL53L1X" library by Pololu. Serial Monitor at 115200.
#include <Wire.h>
#include <VL53L1X.h>

// ---------- Motor pins ----------
#define AIN1 26
#define AIN2 25
#define PWMA 19
#define BIN1 27
#define BIN2 14
#define PWMB 18
#define STBY 13

// ---------- Encoder pins (C1 only) ----------
#define ENC_L_C1 34
#define ENC_R_C1 2

// ---------- Sensor pins ----------
#define XSHUT_FRONT 33
#define XSHUT_LEFT  32
#define XSHUT_RIGHT 4

// ---------- Settings (copy your calibrated values here) ----------
int  speedFwd  = 150;
int  speedTurn = 100;
long turnCounts = 300;          // <-- put your calibrated value here
const int HAND_MM = 100;        // hand closer than this triggers an action
const unsigned long TURN_TIMEOUT = 3000;

const int LEFT_DIR  = -1;
const int RIGHT_DIR = -1;

VL53L1X sFront, sLeft, sRight;
volatile long encL = 0, encR = 0;

void IRAM_ATTR isrL() { encL++; }
void IRAM_ATTR isrR() { encR++; }

// ---------- Motors ----------
void setLeft(int speed) {
  speed *= LEFT_DIR;
  digitalWrite(AIN1, speed > 0);
  digitalWrite(AIN2, speed < 0);
  analogWrite(PWMA, constrain(abs(speed), 0, 255));
}

void setRight(int speed) {
  speed *= RIGHT_DIR;
  digitalWrite(BIN1, speed > 0);
  digitalWrite(BIN2, speed < 0);
  analogWrite(PWMB, constrain(abs(speed), 0, 255));
}

void stopMotors() { setLeft(0); setRight(0); }

void brakeMotors() {
  digitalWrite(AIN1, HIGH); digitalWrite(AIN2, HIGH); analogWrite(PWMA, 255);
  digitalWrite(BIN1, HIGH); digitalWrite(BIN2, HIGH); analogWrite(PWMB, 255);
  delay(80);
  stopMotors();
}

void resetEnc() { noInterrupts(); encL = 0; encR = 0; interrupts(); }
long getL() { noInterrupts(); long v = encL; interrupts(); return v; }
long getR() { noInterrupts(); long v = encR; interrupts(); return v; }

// dir: +1 = right, -1 = left
void turn90(int dir) {
  resetEnc();
  const int  minSpeed = 70;
  const long slowZone = 120;
  unsigned long start = millis();
  bool leftDone = false, rightDone = false;

  while (!(leftDone && rightDone) && millis() - start < TURN_TIMEOUT) {
    if (!leftDone) {
      long remain = turnCounts - getL();
      if (remain <= 0) { setLeft(0); leftDone = true; }
      else {
        int sp = (remain < slowZone) ? map(remain, 0, slowZone, minSpeed, speedTurn) : speedTurn;
        setLeft(dir * sp);
      }
    }
    if (!rightDone) {
      long remain = turnCounts - getR();
      if (remain <= 0) { setRight(0); rightDone = true; }
      else {
        int sp = (remain < slowZone) ? map(remain, 0, slowZone, minSpeed, speedTurn) : speedTurn;
        setRight(-dir * sp);
      }
    }
  }
  brakeMotors();
  delay(150);
  Serial.print("Turn done  L="); Serial.print(getL());
  Serial.print("  R="); Serial.println(getR());
}

// ---------- Sensors ----------
bool startSensor(VL53L1X &s, int xshutPin, uint8_t addr, const char* name) {
  pinMode(xshutPin, INPUT);
  delay(10);
  s.setTimeout(500);
  if (!s.init()) {
    Serial.print("FAILED to init "); Serial.println(name);
    return false;
  }
  s.setAddress(addr);
  s.setDistanceMode(VL53L1X::Short);
  s.setMeasurementTimingBudget(33000);
  s.startContinuous(33);
  Serial.print(name); Serial.println(" OK");
  return true;
}

// returns a large number if the read timed out, so it never counts as a hand
int readMM(VL53L1X &s) {
  int d = s.read();
  if (s.timeoutOccurred()) return 9999;
  return d;
}

// ---------- Setup ----------
void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  Wire.setClock(400000);

  pinMode(AIN1, OUTPUT); pinMode(AIN2, OUTPUT); pinMode(PWMA, OUTPUT);
  pinMode(BIN1, OUTPUT); pinMode(BIN2, OUTPUT); pinMode(PWMB, OUTPUT);
  pinMode(STBY, OUTPUT);
  digitalWrite(STBY, HIGH);
  stopMotors();

  pinMode(ENC_L_C1, INPUT);
  pinMode(ENC_R_C1, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(ENC_L_C1), isrL, CHANGE);
  attachInterrupt(digitalPinToInterrupt(ENC_R_C1), isrR, CHANGE);

  pinMode(XSHUT_FRONT, OUTPUT); digitalWrite(XSHUT_FRONT, LOW);
  pinMode(XSHUT_LEFT,  OUTPUT); digitalWrite(XSHUT_LEFT,  LOW);
  pinMode(XSHUT_RIGHT, OUTPUT); digitalWrite(XSHUT_RIGHT, LOW);
  delay(50);

  startSensor(sFront, XSHUT_FRONT, 0x30, "Front");
  startSensor(sLeft,  XSHUT_LEFT,  0x31, "Left");
  startSensor(sRight, XSHUT_RIGHT, 0x32, "Right");

  Serial.println("Ready: hand on LEFT = left 90 | RIGHT = right 90 | FRONT = forward");
}

// ---------- Loop ----------
bool movingForward = false;
bool waitForRelease = false;   // after a turn, wait until hand is removed

void loop() {
  int f = readMM(sFront);
  int l = readMM(sLeft);
  int r = readMM(sRight);

  bool handF = f < HAND_MM;
  bool handL = l < HAND_MM;
  bool handR = r < HAND_MM;

  // After a turn, wait until the hand is gone from both side sensors
  if (waitForRelease) {
    if (!handL && !handR) waitForRelease = false;
    return;
  }

  if (handL) {
    if (movingForward) { brakeMotors(); movingForward = false; }
    Serial.println("HAND LEFT -> turn left");
    turn90(-1);
    waitForRelease = true;
  }
  else if (handR) {
    if (movingForward) { brakeMotors(); movingForward = false; }
    Serial.println("HAND RIGHT -> turn right");
    turn90(+1);
    waitForRelease = true;
  }
  else if (handF) {
    if (!movingForward) {
      Serial.println("HAND FRONT -> forward");
      setLeft(speedFwd);
      setRight(speedFwd);
      movingForward = true;
    }
  }
  else if (movingForward) {
    Serial.println("HAND REMOVED -> stop");
    brakeMotors();
    movingForward = false;
  }
}
